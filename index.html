GitHub Pages 배포 및 실무 꿀팁

1. GitHub Pages 웹사이트로 무료 배포하기

  1. GitHub에서 새 저장소(Repository) 생성 (예: image-downloader)
  2. 이 index.html 파일을 저장소 루트에 업로드 및 커밋
  3. 저장소의 Settings > Pages 메뉴로 이동
  4. Build and deployment에서 Branch를 main (또는 master) / root 로 선택 후 Save
  5. 1분 뒤 https://사용자명.github.io/image-downloader 웹사이트 자동 완성!

2. 브라우저 보안 제약(CORS) 대응 안내

웹 브라우저는 보안상 다른 사이트의 이미지를 스크립트로 다운로드할 때 CORS 차단이 발생할 수 있습니다. 이미지가 차단되어 다운로드가 안 될 경우 아래 방법 중 하나를 활용하세요:

  • CORS 프록시 옵션 켜기: 아래 설정에서 'CORS 프록시 사용'을 체크하면 프록시 서버를 통해 이미지를 우회 다운로드합니다.
  • 크롬 확장 프로그램: 'Allow CORS: Access-Control-Allow-Origin' 확장 프로그램을 켜면 브라우저 제약이 완전히 풀립니다.
  • 개발자도구 콘솔(F12) 직행: 해당 업무 사이트 화면에서 F12 콘솔에 직접 스크립트를 넣으면 같은 출처로 취급되어 100% 막힘없이 즉시 ZIP 저장이 가능합니다.

작업 사이트에서 바로 실행하는 원클릭 콘솔 스크립트

별도 사이트에 HTML을 복사/붙여넣을 필요 없이, 작업 중인 웹페이지에서 F12(개발자도구) 콘솔에 복사-붙여넣기만 하면 해당 페이지의 <img alt="이미지 N"> 태그들을 즉시 번호순으로 묶어 ZIP으로 다운로드해 주는 코드입니다. (CORS 문제 100% 없음) 

/* 작업 중인 웹페이지의 F12 > Console에 붙여넣고 Enter를 누르세요 */
(async () => {
  if (!window.JSZip) {
    const s = document.createElement('script');
    s.src = 'https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js';
    document.head.appendChild(s);
    await new Promise(r => s.onload = r);
  }
  const imgs = Array.from(document.querySelectorAll('img'))
    .map(img => {
      const alt = img.getAttribute('alt') || '';
      const src = img.currentSrc || img.src;
      const numMatch = alt.match(/\d+/);
      const num = numMatch ? parseInt(numMatch[0], 10) : 99999;
      return { alt, src, num };
    })
    .filter(item => item.src && !item.src.startsWith('data:image/svg'))
    .sort((a, b) => a.num - b.num);

  if (!imgs.length) return alert('이미지를 찾을 수 없습니다.');
  console.log(`총 ${imgs.length}개 이미지 압축 시작...`);
  const zip = new JSZip();
  let count = 0;
  for (let i = 0; i < imgs.length; i++) {
    const { src, num } = imgs[i];
    try {
      const res = await fetch(src);
      const blob = await res.blob();
      const ext = (blob.type.split('/')[1] || 'jpg').replace('jpeg', 'jpg');
      const filename = `이미지_${String(i + 1).padStart(2, '0')}.${ext}`;
      zip.file(filename, blob);
      count++;
      console.log(`[${count}/${imgs.length}] ${filename} 추가 완료`);
    } catch (e) {
      console.error('다운로드 실패:', src, e);
    }
  }
  const content = await zip.generateAsync({ type: 'blob' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(content);
  a.download = '이미지_정리_결과.zip';
  a.click();
  console.log('ZIP 파일 다운로드 완료!');
})();
