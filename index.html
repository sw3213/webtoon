사용 방법 및 GitHub Pages 배포 가이드

1. <div id="toon_img"> 소스 붙여넣기 팁

웹툰 페이지 전체 소스(Ctrl + U)를 그대로 복사해 넣으셔도 되고, 개발자도구(F12)에서 <div id="toon_img"> 태그만 우클릭 > Copy element 하여 붙여넣으셔도 프로그램이 알아서 해당 영역의 웹툰 컷만 골라냅니다.

2. 지연 로딩(data-src) 자동 지원

많은 웹툰 사이트가 스크롤을 내리기 전까지 진짜 이미지 주소를 data-src 또는 data-original에 숨겨둡니다. 이 프로그램은 src뿐만 아니라 지연 로딩 속성까지 모두 검사하여 실제 고화질 원본 주소를 가져옵니다.

3. 브라우저 보안(CORS) 자동 우회

외부 이미지 서버가 직접 다운로드를 막을 경우를 대비해, 다운로드 실패 시 공개 CORS 프록시 서버로 2차 자동 재시도하여 ZIP으로 묶어줍니다.

#toon_img 전용 F12 콘솔 스크립트

웹툰 페이지에서 F12 > Console에 붙여넣으면, 페이지의 <div id="toon_img"> 안의 이미지를 1번부터 차례대로 ZIP 압축하여 내려받습니다. 

/* 웹툰 페이지의 F12 > Console에 붙여넣고 Enter를 누르세요 */
(async () => {
  if (!window.JSZip) {
    const s = document.createElement('script');
    s.src = 'https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js';
    document.head.appendChild(s);
    await new Promise(r => s.onload = r);
  }
  const container = document.getElementById('toon_img') || document;
  const imgs = Array.from(container.querySelectorAll('img'))
    .map((img, idx) => {
      const alt = img.getAttribute('alt') || '';
      const src = img.getAttribute('data-src') || img.getAttribute('data-original') || img.getAttribute('data-url') || img.currentSrc || img.src;
      const numMatch = alt.match(/\d+/);
      const num = numMatch ? parseInt(numMatch[0], 10) : (idx + 1);
      return { alt, src, num, originalIdx: idx };
    })
    .filter(item => item.src && !item.src.startsWith('data:image/svg') && !item.src.includes('spacer'))
    .sort((a, b) => a.num - b.num);

  if (!imgs.length) return alert('#toon_img 안에서 이미지를 찾을 수 없습니다.');
  console.log(`[toon_img] 총 ${imgs.length}개 이미지 압축 시작...`);
  const zip = new JSZip();
  let count = 0;
  for (let i = 0; i < imgs.length; i++) {
    const { src } = imgs[i];
    try {
      const res = await fetch(src);
      const blob = await res.blob();
      const ext = (blob.type.split('/')[1] || 'jpg').replace('jpeg', 'jpg');
      const filename = `이미지_${String(i + 1).padStart(2, '0')}.${ext}`;
      zip.file(filename, blob);
      count++;
      console.log(`[${count}/${imgs.length}] ${filename} 추가 완료`);
    } catch (e) {
      console.error('다운로드 실패:', src, e);
    }
  }
  const content = await zip.generateAsync({ type: 'blob' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(content);
  a.download = '웹툰_toon_img_정리.zip';
  a.click();
  console.log('ZIP 파일 다운로드 완료!');
})();
